## 本页目标

从"颜色"换到"标记"：二维码自带 ID，能区分**这是哪个站台**。本页先学会生成二维码卡片，再用 `cv2.QRCodeDetector().detectAndDecode()` 一次拿到文字 + 四角坐标，进而算出中心与偏差。

## 前置条件

- 完成 4.6（会算归一化偏差）
- 同目录有 `cards/` 文件夹（放测试图片，可自己拍照替换）

## 动手步骤

**1. 新建 `27_qr_basics.py`**

**2. 写入核心代码**

```python
from pathlib import Path
import cv2
import numpy as np

QR_TEXT = "GREETING"
QR_SCALE = 32                                   # 放大倍数，约 900 像素，手机屏幕也够看
TEST_IMAGE_NAME = "01_GREETING_LEFT_TOP.png"
OUTPUT_DIR = Path(__file__).parent / "cards"
OUTPUT_PATH = OUTPUT_DIR / f"{QR_TEXT}.png"
TEST_IMAGE_PATH = OUTPUT_DIR / TEST_IMAGE_NAME


def create_qr(text):
    encoder = cv2.QRCodeEncoder_create()
    qr_image = encoder.encode(text)
    qr_image = cv2.copyMakeBorder(qr_image, 4, 4, 4, 4, cv2.BORDER_CONSTANT, value=255)  # 白边
    qr_image = cv2.resize(qr_image, None, fx=QR_SCALE, fy=QR_SCALE,
                          interpolation=cv2.INTER_NEAREST)   # 最近邻，不留灰过渡
    OUTPUT_DIR.mkdir(exist_ok=True)
    cv2.imencode(".png", qr_image)[1].tofile(OUTPUT_PATH)


def load_image(path):
    return cv2.imdecode(np.fromfile(path, dtype=np.uint8), cv2.IMREAD_COLOR)


def find_qr(image):
    text, points, _ = cv2.QRCodeDetector().detectAndDecode(image)
    if not text or points is None:
        return {"text": None, "points": None, "center": None,
                "area_ratio": 0.0, "error_x": None, "error_y": None}

    corners = points.reshape(4, 2)                       # (1,4,2) → 4 行 (x, y)
    center = corners.mean(axis=0)                        # 四角平均 = 中心
    center_x, center_y = round(center[0]), round(center[1])
    height, width = image.shape[:2]
    area_ratio = abs(cv2.contourArea(corners.astype(np.float32))) / (width * height)

    return {"text": text, "points": corners, "center": (center_x, center_y),
            "area_ratio": area_ratio,
            "error_x": (center_x - width / 2) / (width / 2),    # 左负右正
            "error_y": (center_y - height / 2) / (height / 2)}  # 上负下正


def draw_qr_marks(image, qr_data):
    marked_image = image.copy()
    height, width = image.shape[:2]
    image_center = (round(width / 2), round(height / 2))
    if qr_data["points"] is not None:
        corners = qr_data["points"].round().astype(int)
        cv2.polylines(marked_image, [corners], True, (255, 200, 0), 4)   # 闭合四边形
        cv2.circle(marked_image, qr_data["center"], 9, (0, 0, 255), -1)
        cv2.arrowedLine(marked_image, image_center, qr_data["center"], (0, 176, 255), 3)
    text = f"QR={qr_data['text']}  X={qr_data['error_x']}  Y={qr_data['error_y']}"
    cv2.rectangle(marked_image, (15, 15), (720, 70), (32, 32, 32), -1)
    cv2.putText(marked_image, text, (30, 52), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 176, 255), 2)
    return marked_image


def main():
    create_qr(QR_TEXT)
    test_image = load_image(TEST_IMAGE_PATH)
    qr_data = find_qr(test_image)
    print(f"识别文字: {qr_data['text']}")
    print(f"中心坐标: {qr_data['center']}")
    print(f"二维偏差: X={qr_data['error_x']:.3f}, Y={qr_data['error_y']:.3f}")
    cv2.imshow("QR basics - press any key to close", draw_qr_marks(test_image, qr_data))
    cv2.waitKey(0)
    cv2.destroyAllWindows()


if __name__ == "__main__":
    main()
```

**3. 运行，`cards/GREETING.png` 会生成**，把它显示在手机或打印出来对准摄像头试

## 参数说明

| 参数 | 类型 | 默认值或范围 | 说明 |
| --- | --- | --- | --- |
| `QR_TEXT` | str | 任意短文本 | 写入二维码的内容，就是后面的"站台 ID" |
| `QR_SCALE` | int | 20～40 | 太小手机显示不清，太大占内存 |
| `INTER_NEAREST` | 常量 | — | 必须最近邻，双线性会让黑白边界发灰导致识别失败 |
| `area_ratio` | float | 0～1 | 二维码占画面面积比，可当"距离远近代餐" |

## 数据结构

`find_qr()` 返回值（识别失败时字段一致，只是值为 `None` / `0.0`）：

```python
{
    "text": "GREETING",            # 解码出的文字；失败为 None
    "points": [[x1,y1],[x2,y2],[x3,y3],[x4,y4]],   # 四角坐标 (4,2) 数组
    "center": (321, 240),          # 四角平均中心；失败为 None
    "area_ratio": 0.087,           # 面积占比；失败为 0.0
    "error_x": -0.21,              # 归一化横向偏差；失败为 None
    "error_y": -0.34,              # 归一化纵向偏差；失败为 None
}
```

## 预期效果与排查

窗口显示二维码被浅蓝四边形框住、红点在中心、橘黄箭头从画面中心指向二维码。

| 现象 | 原因 | 处理 |
| --- | --- | --- |
| `text` 一直是 `None` | 图太小 / 模糊 / 没有白边 | 提高 `QR_SCALE`，拍照时留白边、别逆光 |
| 四角顺序看起来乱 | 正常，`detectAndDecode` 按检测顺序返回 | 只用来画框和算中心，不要假设哪个角是左上 |
| 中文内容解不出 | 编码问题 | 用英文/数字做站点 ID 最稳 |
| 生成图全黑 | 忘了白边或值域错 | `copyMakeBorder` 的 `value=255` 不能省 |
