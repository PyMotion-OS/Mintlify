## 本页目标

把 4.5 的"找到轮廓"变成一句能给控制用的结论：**目标在画面哪个位置、偏了多少、有多大**。核心是用 `cv2.moments()` 算质心，再把像素坐标换算成归一化偏差 `error_x` / `error_y`。

## 前置条件

- 同目录有 `camera_frame.jpg`、`target_contours.py`（4.5 抽出的函数）
- 完成 4.5

## 动手步骤

**1. 新建 `21_color_target_observation.py`**

**2. 写入核心代码**

```python
from pathlib import Path
import cv2
import numpy as np
from target_contours import find_target_contours

TARGET_TYPE = "blue"
CENTER_X = 0.50      # 参考中心在画面宽度 50%
CENTER_Y = 0.50      # 参考中心在画面高度 50%

BASE_DIR = Path(__file__).parent
IMAGE_PATH = BASE_DIR / "camera_frame.jpg"


def load_image(path):
    return cv2.imdecode(np.fromfile(path, dtype=np.uint8), cv2.IMREAD_COLOR)


def center_to_error(center, image_shape):
    height, width = image_shape[:2]
    reference_x = round(width * CENTER_X)
    reference_y = round(height * CENTER_Y)
    error_x = (center[0] - reference_x) / (width / 2)     # 左负右正，范围约 -1～+1
    error_y = (center[1] - reference_y) / (height / 2)    # 上负下正，范围约 -1～+1
    return error_x, error_y


def find_target(image, target_type="blue"):
    target_mask, valid_contours, roi = find_target_contours(image, target_type)
    left, top, right, bottom = roi

    if not valid_contours:
        return {"state": "lost", "center": None, "area": 0,
                "error_x": None, "error_y": None, "mask": target_mask,
                "contour": None, "roi": roi}

    target_contour = max(valid_contours, key=cv2.contourArea)
    moments = cv2.moments(target_contour)
    center_x = left + round(moments["m10"] / moments["m00"])   # 横向累计 / 面积
    center_y = top + round(moments["m01"] / moments["m00"])    # 纵向累计 / 面积
    error_x, error_y = center_to_error((center_x, center_y), image.shape)

    return {"state": "target", "center": (center_x, center_y),
            "area": round(moments["m00"]),
            "error_x": error_x, "error_y": error_y,
            "mask": target_mask, "contour": target_contour, "roi": roi}


def draw_target_marks(image, target_data):
    marked_image = image.copy()
    reference_x = round(image.shape[1] * CENTER_X)
    reference_y = round(image.shape[0] * CENTER_Y)
    left, top, right, bottom = target_data["roi"]

    cv2.rectangle(marked_image, (left, top), (right - 1, bottom - 1), (217, 199, 0), 4)
    cv2.line(marked_image, (reference_x, top), (reference_x, bottom - 1), (255, 255, 255), 3)
    cv2.line(marked_image, (left, reference_y), (right - 1, reference_y), (255, 255, 255), 3)

    if target_data["state"] == "target":
        cv2.drawContours(marked_image, [target_data["contour"]], -1,
                         (0, 140, 255), 4, offset=(left, top))   # offset 把 ROI 坐标平移回原图
        cv2.circle(marked_image, target_data["center"], 10, (0, 0, 255), -1)
        cv2.arrowedLine(marked_image, (reference_x, reference_y),
                        target_data["center"], (0, 176, 255), 4)
    return marked_image


def main():
    image = load_image(IMAGE_PATH)
    target_data = find_target(image, TARGET_TYPE)
    print(f"目标状态: {target_data['state']}")
    print(f"目标中心: {target_data['center']}")
    print(f"目标面积: {target_data['area']}")
    print(f"横向偏差: {target_data['error_x']}")
    print(f"纵向偏差: {target_data['error_y']}")

    cv2.imshow("Target Observation", draw_target_marks(image, target_data))
    cv2.waitKey(0)
    cv2.destroyAllWindows()


if __name__ == "__main__":
    main()
```

**3. 运行，看白线交点与红点的关系**

## 参数说明

| 参数 | 类型 | 默认值或范围 | 说明 |
| --- | --- | --- | --- |
| `TARGET_TYPE` | str | `"blue"` | 目标筛选方式 |
| `CENTER_X` / `CENTER_Y` | float | 0.0～1.0 | 参考中心在画面中的比例位置，十字白线就画在这里 |
| `error_x` | float | 约 -1～+1 | 负=目标在参考点左侧，正=右侧 |
| `error_y` | float | 约 -1～+1 | 负=目标在参考点上方，正=下方 |
| `area` | int | 像素数 | `m00` 即轮廓面积，后续用来判断距离远近 |

## 数据结构

`find_target()` 返回的字典（`lost` 与 `target` 两种状态下**字段完全一致**，这样下游不用做分支判断）：

```python
{
    "state": "target",       # "target" 找到目标 / "lost" 未找到
    "center": (321, 240),    # 原图坐标 (x, y)；lost 时为 None
    "area": 5120,            # 目标像素面积；lost 时为 0
    "error_x": 0.08,         # 横向归一化偏差；lost 时为 None
    "error_y": -0.13,        # 纵向归一化偏差；lost 时为 None
    "mask": <ndarray>,       # 清理后的单通道黑白掩膜
    "contour": <ndarray>,    # ROI 内目标轮廓（绘制时要带 offset）；lost 时为 None
    "roi": (96, 72, 544, 408)  # 处理区域 (left, top, right, bottom)
}
```

## 预期效果与排查

白十字交点是参考中心，红点是目标质心，橘黄箭头从参考点指向目标。目标居中时 `error_x`、`error_y` 都接近 0。

| 现象 | 原因 | 处理 |
| --- | --- | --- |
| 中心点跑到目标外面 | 轮廓不闭合 / 掩膜破碎 | 回 4.5 加大闭运算核 |
| `error` 超出 ±1 | 目标在画面边缘 | 正常现象，说明目标已偏出参考区 |
| 总有 `state: lost` | 阈值不动或场景里没这个颜色 | 换 `TARGET_TYPE` 或换素材图 |
| 画出的轮廓位置整体偏移 | 少了 `offset=(left, top)` | 轮廓坐标是 ROI 内坐标，绘制必须带偏移 |
